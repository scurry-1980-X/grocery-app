      <input id="name" placeholder="e.g. Apples"

             maxlength="150" required autocomplete="off">

    </label>

    <label>Quantity

      <input id="quantity" type="number" value="1"

             min="1" max="999" step="1" required>

    </label>

    <button>Add to list</button>

  </form>


  <p id="status" role="status"></p>

  <ul id="list" aria-label="Groceries"></ul>

  <button id="clear" type="button">Clear purchased items</button>

  <p class="note">

    Your list stays in this browser on this device.

    Clearing browser data will erase it.

  </p>

<script>

const key = 'my-groceries-v1';

const list = document.querySelector('#list');

const status = document.querySelector('#status');

const nameInput = document.querySelector('#name');

const quantityInput = document.querySelector('#quantity');

let items = [];

let storageAvailable = true;


try {

  const saved = JSON.parse(localStorage.getItem(key) || '[]');

  if (Array.isArray(saved)) {

    items = saved.filter(item =>

      item && typeof item.name === 'string' &&

      Number.isInteger(item.quantity) &&

      item.quantity >= 1 && item.quantity <= 999 &&

      typeof item.done === 'boolean'

    );

  }

} catch {

  storageAvailable = false;

}


function save() {

  try {

    localStorage.setItem(key, JSON.stringify(items));

    storageAvailable = true;

  } catch {

    storageAvailable = false;

  }

  render();

}


function render() {

  list.replaceChildren();

  const remaining = items.filter(item => !item.done).length;

  status.textContent = items.length

    ? `${remaining} of ${items.length} items left to buy`

    : 'Your list is empty. Add your first item above.';

  if (!storageAvailable) {

    status.textContent +=

      ' Browser storage is unavailable; this list may not survive closing the page.';

  }


  items.forEach((item, index) => {

    const row = document.createElement('li');

    row.className = item.done ? 'done' : '';


    const check = document.createElement('input');

    check.type = 'checkbox';

    check.checked = item.done;

    check.setAttribute('aria-label', `Purchased ${item.name}`);

    check.addEventListener('change', () => {

      item.done = check.checked;

      save();

      const replacement = list.children[index];

      if (replacement) replacement.querySelector('input').focus();

    });


    const text = document.createElement('span');

    text.className = 'item';

    text.textContent = `${item.name} × ${item.quantity}`;


    const remove = document.createElement('button');

    remove.type = 'button';

    remove.className = 'remove';

    remove.textContent = '✕';

    remove.setAttribute('aria-label', `Remove ${item.name}`);

    remove.addEventListener('click', () => {

      items.splice(index, 1);

      save();

      const next = list.children[Math.min(index, items.length - 1)];

      if (next) next.querySelector('.remove').focus();

      else nameInput.focus();

    });


    row.append(check, text, remove);

    list.append(row);

  });

  document.querySelector('#clear').disabled =

    !items.some(item => item.done);

}


document.querySelector('#form').addEventListener('submit', event => {

  event.preventDefault();

  const name = nameInput.value.trim();

  if (!name) {

    nameInput.focus();

    return;

  }

  items.push({

    name,

    quantity: Number(quantityInput.value),

    done: false

  });

  save();

  nameInput.value = '';

  quantityInput.value = '1';

  nameInput.focus();

});


document.querySelector('#clear').addEventListener('click', () => {

  items = items.filter(item => !item.done);

  save();

  nameInput.focus();

});


render();

</script>
